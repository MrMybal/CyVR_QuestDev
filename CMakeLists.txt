cmake_minimum_required(VERSION 3.20)
project(CyVR_QuestDev VERSION 0.5.0 LANGUAGES CXX)
# CMake can misdecode French MSVC's UTF-8 /showIncludes prefix as OEM text.
# Keep Ninja's header dependency scanner aligned with the compiler output.
if(MSVC AND CMAKE_CL_SHOWINCLUDES_PREFIX MATCHES "^Remarque")
    set(CMAKE_CL_SHOWINCLUDES_PREFIX "Remarque : inclusion du fichier : ")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
    add_compile_options(/EHsc)
endif()
if(NOT WIN32)
    message(FATAL_ERROR "CyVR QuestDev requires Windows (Win32 / DirectX 11).")
endif()
set(CYUI_ROOT "$ENV{CYUI_ROOT}" CACHE PATH "CyImgui_Interface source directory")
if(NOT CYUI_ROOT)
    foreach(parent ".." "../.." "../../..")
        get_filename_component(candidate "${CMAKE_SOURCE_DIR}/${parent}/CyImgui_Interface" ABSOLUTE)
        if(EXISTS "${candidate}/include/cyui/CyUI.h")
            set(CYUI_ROOT "${candidate}" CACHE PATH "CyImgui_Interface source directory" FORCE)
            break()
        endif()
    endforeach()
endif()
if(NOT EXISTS "${CYUI_ROOT}/CMakeLists.txt")
    message(FATAL_ERROR "Set CYUI_ROOT to the CyImgui_Interface source directory.")
endif()
set(CYUI_BUILD_DEMO OFF CACHE BOOL "" FORCE)
add_subdirectory("${CYUI_ROOT}" "${CMAKE_BINARY_DIR}/cyui")
set(IMGUI_DIR "${CYUI_ROOT}/third_party/imgui")
add_library(quest_core STATIC src/core.cpp)
target_include_directories(quest_core PUBLIC include)
add_library(quest_backend STATIC src/backend.cpp src/plugins.cpp src/metrics_plugins.cpp src/files.cpp src/metrics_data.cpp src/metrics_capture.cpp)
target_link_libraries(quest_backend PUBLIC quest_core)
target_include_directories(quest_backend PUBLIC include third_party)
add_executable(CyVR_QuestDev WIN32 src/main.cpp src/app.cpp src/preview.cpp src/assistant_bridge.cpp src/metrics_view.cpp
    "${IMGUI_DIR}/backends/imgui_impl_win32.cpp"
    "${IMGUI_DIR}/backends/imgui_impl_dx11.cpp")
target_include_directories(CyVR_QuestDev PRIVATE "${IMGUI_DIR}/backends" third_party)
target_link_libraries(CyVR_QuestDev PRIVATE quest_backend CyUI::CyUI d3d11 dxgi d3dcompiler dwmapi shell32 comdlg32 ole32 windowscodecs)
add_custom_command(TARGET CyVR_QuestDev POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_SOURCE_DIR}/plugins" "$<TARGET_FILE_DIR:CyVR_QuestDev>/plugins")
target_compile_definitions(CyVR_QuestDev PRIVATE UNICODE _UNICODE NOMINMAX)
if(MSVC)
    foreach(target quest_core quest_backend CyVR_QuestDev)
        target_compile_options(${target} PRIVATE /W4 /utf-8 /EHsc)
    endforeach()
endif()
include(CTest)
if(BUILD_TESTING)
    add_executable(quest_tests tests/core_tests.cpp)
    target_link_libraries(quest_tests PRIVATE quest_backend)
    add_executable(fake_adb tests/fake_adb.cpp)
    target_include_directories(fake_adb PRIVATE third_party)
    add_executable(quest_plugin_tests tests/metrics_plugin_tests.cpp)
    add_executable(quest_metrics_tests tests/metrics_tests.cpp)
    target_link_libraries(quest_metrics_tests PRIVATE quest_backend)
    target_link_libraries(quest_plugin_tests PRIVATE quest_backend)
    add_executable(quest_file_tests tests/files_tests.cpp)
    target_link_libraries(quest_file_tests PRIVATE quest_backend)
    if(MSVC)
        target_compile_options(quest_tests PRIVATE /W4 /utf-8 /EHsc)
        target_compile_options(fake_adb PRIVATE /W4 /utf-8 /EHsc)
        target_compile_options(quest_plugin_tests PRIVATE /W4 /utf-8 /EHsc)
        target_compile_options(quest_metrics_tests PRIVATE /W4 /utf-8 /EHsc)
        target_compile_options(quest_file_tests PRIVATE /W4 /utf-8 /EHsc)
    endif()
    add_test(NAME quest_core_and_backend COMMAND quest_tests $<TARGET_FILE:fake_adb>)
    add_test(NAME quest_metrics_plugins COMMAND quest_plugin_tests --fake-adb $<TARGET_FILE:fake_adb>)
    add_test(NAME quest_csv_and_metrics COMMAND quest_metrics_tests $<TARGET_FILE:fake_adb>)
    find_program(QUEST_TEST_SH NAMES sh PATHS "C:/Program Files/Git/usr/bin" NO_DEFAULT_PATH)
    if(QUEST_TEST_SH)
        add_test(NAME quest_file_operations COMMAND quest_file_tests "${QUEST_TEST_SH}")
    else()
        add_test(NAME quest_file_operations COMMAND quest_file_tests)
    endif()
endif()
