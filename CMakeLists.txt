cmake_minimum_required(VERSION 3.20)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" QUESTDEV_RELEASE_VERSION LIMIT_COUNT 1)
string(REGEX MATCH "^[0-9]+\\.[0-9]+\\.[0-9]+" QUESTDEV_CORE_VERSION "${QUESTDEV_RELEASE_VERSION}")
project(CyVR_QuestDev VERSION ${QUESTDEV_CORE_VERSION} LANGUAGES CXX)
configure_file(include/quest/version.hpp.in "${CMAKE_BINARY_DIR}/generated/quest/version.hpp" @ONLY)
# CMake can misdecode French MSVC's UTF-8 /showIncludes prefix as OEM text.
# Keep Ninja's header dependency scanner aligned with the compiler output.
if(MSVC AND CMAKE_CL_SHOWINCLUDES_PREFIX MATCHES "^Remarque")
    set(CMAKE_CL_SHOWINCLUDES_PREFIX "Remarque : inclusion du fichier : ")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
if(MSVC)
    add_compile_options(/EHsc)
endif()
if(NOT WIN32)
    message(FATAL_ERROR "CyVR QuestDev requires Windows (Win32 / DirectX 11).")
endif()
enable_language(RC)
configure_file(src/resources.rc.in "${CMAKE_BINARY_DIR}/quest_resources.rc" @ONLY)
include(cmake/CyUI.cmake)
add_library(quest_core STATIC src/core.cpp src/documentation.cpp)
target_include_directories(quest_core PUBLIC include PRIVATE third_party)
add_library(quest_backend STATIC src/backend.cpp src/plugins.cpp src/metrics_plugins.cpp src/files.cpp src/transfers.cpp src/metrics_data.cpp src/metrics_capture.cpp src/mirroring.cpp)
target_link_libraries(quest_backend PUBLIC quest_core user32)
target_include_directories(quest_backend PUBLIC include third_party "${CMAKE_BINARY_DIR}/generated")
add_library(quest_text_views STATIC src/log_view.cpp)
target_link_libraries(quest_text_views PUBLIC quest_core CyUI::Core PRIVATE CyUI::CodeEditor)
target_compile_features(quest_text_views PRIVATE cxx_std_20)
target_include_directories(quest_text_views PUBLIC include)
if(MSVC)
    target_compile_options(quest_text_views PRIVATE /W4 /utf-8 /EHsc)
endif()
add_library(quest_plot_views STATIC src/metrics_presentation.cpp)
target_link_libraries(quest_plot_views PUBLIC quest_backend CyUI::Plots)
add_executable(CyVR_QuestDev WIN32 src/main.cpp src/app.cpp src/ui_locale.cpp src/preview.cpp src/assistant_bridge.cpp src/metrics_view.cpp src/file_workspace.cpp
    "${CMAKE_BINARY_DIR}/quest_resources.rc"
    "${IMGUI_DIR}/backends/imgui_impl_win32.cpp"
    "${IMGUI_DIR}/backends/imgui_impl_dx11.cpp")
target_include_directories(CyVR_QuestDev PRIVATE "${IMGUI_DIR}/backends" third_party)
target_link_libraries(CyVR_QuestDev PRIVATE quest_backend quest_text_views quest_plot_views CyUI::Core CyUI::Plots CyUI::Images CyUI::DocumentViews d3d11 dxgi d3dcompiler dwmapi shell32 comdlg32 ole32 windowscodecs)
add_custom_command(TARGET CyVR_QuestDev POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_SOURCE_DIR}/plugins" "$<TARGET_FILE_DIR:CyVR_QuestDev>/plugins")
if(EXISTS "${CMAKE_SOURCE_DIR}/Documentation/Documentation.html")
    add_custom_target(quest_documentation ALL
        COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_SOURCE_DIR}/Documentation" "$<TARGET_FILE_DIR:CyVR_QuestDev>/Documentation"
        DEPENDS CyVR_QuestDev
        COMMENT "Copy the complete QuestDev manual")
else()
    message(WARNING "Generate the manual with python tools/documentation.py before distributing QuestDev.")
endif()
set_property(TARGET CyVR_QuestDev APPEND PROPERTY LINK_DEPENDS
    "${CMAKE_SOURCE_DIR}/LICENSE" "${CMAKE_SOURCE_DIR}/COPYRIGHT" "${CMAKE_SOURCE_DIR}/THIRD_PARTY.md"
    "${CMAKE_SOURCE_DIR}/assets/branding/cyvr-questdev-logo-gradient.png"
    "${CMAKE_SOURCE_DIR}/assets/branding/cyvr-questdev.ico")
set_property(SOURCE "${CMAKE_BINARY_DIR}/quest_resources.rc" APPEND PROPERTY OBJECT_DEPENDS
    "${CMAKE_SOURCE_DIR}/assets/branding/cyvr-questdev-logo-gradient.png"
    "${CMAKE_SOURCE_DIR}/assets/branding/cyvr-questdev.ico")
add_custom_command(TARGET CyVR_QuestDev POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_SOURCE_DIR}/LICENSE" "$<TARGET_FILE_DIR:CyVR_QuestDev>/LICENSE.txt"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_SOURCE_DIR}/COPYRIGHT" "$<TARGET_FILE_DIR:CyVR_QuestDev>/COPYRIGHT.txt"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_SOURCE_DIR}/THIRD_PARTY.md" "$<TARGET_FILE_DIR:CyVR_QuestDev>/THIRD_PARTY.md"
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_SOURCE_DIR}/LICENCES" "$<TARGET_FILE_DIR:CyVR_QuestDev>/LICENCES"
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_SOURCE_DIR}/licenses" "$<TARGET_FILE_DIR:CyVR_QuestDev>/licenses")
target_compile_definitions(CyVR_QuestDev PRIVATE UNICODE _UNICODE NOMINMAX)
if(MSVC)
    foreach(target quest_core quest_backend CyVR_QuestDev)
        target_compile_options(${target} PRIVATE /W4 /utf-8 /EHsc)
    endforeach()
endif()
include(CTest)
if(BUILD_TESTING)
    add_executable(quest_documentation_tests tests/documentation_tests.cpp)
    target_link_libraries(quest_documentation_tests PRIVATE quest_core)
    if(MSVC)
        target_compile_options(quest_documentation_tests PRIVATE /W4 /utf-8 /EHsc)
    endif()
    add_test(NAME quest_documentation_links COMMAND quest_documentation_tests)
    add_executable(quest_module_views_tests tests/module_views_tests.cpp src/ui_locale.cpp)
    target_link_libraries(quest_module_views_tests PRIVATE quest_plot_views CyUI::Images CyUI::DocumentViews)
    if(MSVC)
        target_compile_options(quest_module_views_tests PRIVATE /W4 /utf-8 /EHsc)

    add_executable(quest_log_view_tests tests/log_view_tests.cpp)
    target_link_libraries(quest_log_view_tests PRIVATE quest_text_views CyUI::Core)
    if(MSVC)
        target_compile_options(quest_log_view_tests PRIVATE /W4 /utf-8 /EHsc)
    endif()
    add_test(NAME quest_log_viewports COMMAND quest_log_view_tests)
    add_executable(fake_update_app tests/fake_update_app.cpp)
    if(MSVC)
        target_compile_options(fake_update_app PRIVATE /W4 /utf-8 /EHsc)
    endif()
    add_executable(fake_scrcpy tests/fake_scrcpy.cpp)
    add_executable(quest_mirroring_tests tests/mirroring_tests.cpp)
    target_link_libraries(quest_mirroring_tests PRIVATE quest_backend)
    if(MSVC)
        target_compile_options(fake_scrcpy PRIVATE /W4 /utf-8 /EHsc)
        target_compile_options(quest_mirroring_tests PRIVATE /W4 /utf-8 /EHsc)
    endif()
    add_test(NAME quest_video_mirroring COMMAND quest_mirroring_tests $<TARGET_FILE:fake_scrcpy> $<TARGET_FILE:fake_adb>)

    target_link_libraries(quest_tests PRIVATE quest_backend)

    target_link_libraries(quest_plugin_tests PRIVATE quest_backend)

    if(MSVC)

    target_link_libraries(quest_file_tests PRIVATE quest_backend)
    if(MSVC)
        target_compile_options(quest_tests PRIVATE /W4 /utf-8 /EHsc)
        target_compile_options(fake_adb PRIVATE /W4 /utf-8 /EHsc)
        target_compile_options(quest_plugin_tests PRIVATE /W4 /utf-8 /EHsc)
        target_compile_options(quest_metrics_tests PRIVATE /W4 /utf-8 /EHsc)
        target_compile_options(quest_file_tests PRIVATE /W4 /utf-8 /EHsc)
    endif()
    add_test(NAME quest_core_and_backend COMMAND quest_tests $<TARGET_FILE:fake_adb>)
    add_test(NAME quest_metrics_plugins COMMAND quest_plugin_tests --fake-adb $<TARGET_FILE:fake_adb>)
    add_test(NAME quest_csv_and_metrics COMMAND quest_metrics_tests $<TARGET_FILE:fake_adb>)
    find_program(QUEST_TEST_SH NAMES sh PATHS "C:/Program Files/Git/usr/bin" NO_DEFAULT_PATH)
    if(QUEST_TEST_SH)
        add_test(NAME quest_metrics_report_shell COMMAND quest_metrics_tests $<TARGET_FILE:fake_adb> "${QUEST_TEST_SH}")
        add_test(NAME quest_file_operations COMMAND quest_file_tests "${QUEST_TEST_SH}")
    else()
        add_test(NAME quest_file_operations COMMAND quest_file_tests)
    endif()
endif()
